<scoot-warning>

De zonewaarschuwing als losse, zelfstandige web component. Shadow DOM, eigen styling, nul dependencies — werkt in vanilla, React (gewoon als element renderen) of waar dan ook. Dit is de volledige demo-broncode.

Statisch in je markup

Imperatief (met auto-dismiss + events)

Gebruik

// 1. statisch (ook in React/JSX):
import './warning-card.js';
<scoot-warning variant="nadert" distance="80"
  zone="Voetgangersgebied Steenweg" />
// events: onScoot-action / addEventListener('scoot-action')

// 2. imperatief, verschijnt bovenin beeld:
import { showWarning } from './warning-card.js';
showWarning({
  variant: 'verboden',        // verboden | rijbaan | venstertijd | nadert | geen-route
  zone: 'Voetgangersgebied Steenweg',
  distance: 80,               // alleen 'nadert'
  window: 'ma-za 11:00-18:00',// alleen 'venstertijd'
  duration: 9000,             // ms, 0 = blijft staan
  onAction: () => herbereken(),
  onDismiss: () => {},
});

// copy overschrijven kan altijd:
showWarning({ variant: 'verboden', title: '...', body: '...', cta: '...' });
